Google Drive Poster Storage Bridge

Active Drive Bridge

Union Design Hub stores posters, proofs, and references in Google Drive via Google Apps Script.

Paste your deployed Web App URL (Execute as: Me, Access: Anyone).
Root folder where Posters/, Proof/, and Reference/ subfolders reside.

Ready-to-Deploy Google Apps Script Code (Code.gs):

Create a project at script.google.com, paste the code below into Code.gs, click Deploy > New deployment > Web app, set Who has access: Anyone, and paste the URL above! 

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(15000);
  try {
    var data = JSON.parse(e.postData.contents);
    var base64 = data.base64;
    if (base64.indexOf(',') > -1) {
      base64 = base64.split(',')[1];
    }
    var decoded = Utilities.base64Decode(base64);
    var mimeType = data.mimeType || "application/octet-stream";
    var fileName = data.fileName || ("file_" + new Date().getTime());
    var blob = Utilities.newBlob(decoded, mimeType, fileName);

    var rootFolderId = data.rootFolderId || "bc1p7wsvzgq4lp23jmn568n5lamzx5qy9uwuj2t533cda38p56xa35eq9r8rn7";
    var rootFolder;
    try {
      rootFolder = DriveApp.getFolderById(rootFolderId);
    } catch(err) {
      rootFolder = DriveApp.getRootFolder();
    }

    var targetFolder = rootFolder;
    var folderType = data.folderType || "Posters"; // Posters, Proof, Reference
    var language = data.language || "Arabic";
    var committeeName = data.committeeName || "General";

    // 1. Get or create type folder (Posters / Proof / Reference)
    targetFolder = getOrCreateSubFolder(targetFolder, folderType);

    // 2. Posters & Proof have Language / Committee hierarchy
    if (folderType === "Posters" || folderType === "Proof") {
      targetFolder = getOrCreateSubFolder(targetFolder, language);
      targetFolder = getOrCreateSubFolder(targetFolder, committeeName);
    }

    var file = targetFolder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    var fileId = file.getId();
    var response = {
      status: "success",
      fileId: fileId,
      fileName: fileName,
      fileUrl: "https://lh3.googleusercontent.com/d/" + fileId,
      webViewLink: file.getUrl(),
      downloadUrl: "https://drive.google.com/uc?export=download&id=" + fileId,
      folderId: targetFolder.getId(),
      folderPath: folderType + "/" + language + "/" + committeeName,
      fileSize: file.getSize()
    };

    return ContentService.createTextOutput(JSON.stringify(response))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateSubFolder(parentFolder, subFolderName) {
  var folders = parentFolder.getFoldersByName(subFolderName);
  if (folders.hasNext()) {
    return folders.next();
  }
  return parentFolder.createFolder(subFolderName);
}
